import re

def clean_text(text):
    text = text.lower()                                    # lowercase
    text = re.sub(r"http\S+|www\S+", " URL ", text)        # replace urls
    text = re.sub(r"\d+", " NUM ", text)                   # replace numbers
    text = re.sub(r"[^\w\s]", "", text)                    # remove punctuation

def extract_features(texts):
    features = []
    for t in texts:
        doc = nlp(t)

        # Sentiment features
        sent_scores = sia.polarity_scores(t)

        # POS counts
        noun_count = sum(1 for token in doc if token.pos_ == "NOUN")
        verb_count = sum(1 for token in doc if token.pos_ == "VERB")
        adj_count  = sum(1 for token in doc if token.pos_ == "ADJ")
        pron_count = sum(1 for token in doc if token.pos_ == "PRON")

        # Collect features
        features.append({
            "clean_text": t,
            "text_len": len(t.split()),
            "word_count": len(t.split()),
            "char_count": len(t),
            "sent_neg": sent_scores["neg"],
            "sent_neu": sent_scores["neu"],
            "sent_pos": sent_scores["pos"],
            "sent_compound": sent_scores["compound"],
            "noun_count": noun_count,
            "verb_count": verb_count,
            "adj_count": adj_count,
            "pron_count": pron_count
        })

    return pd.DataFrame(features)
